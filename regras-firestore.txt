rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ===== Inscrições de Eventos — versão para vários organizadores =====

    // Administrador geral (painel admin/)
    function logado() { return request.auth != null && request.auth.token.email != null; }
    function meuEmail() { return request.auth.token.email; }
    function isSuper() { return logado() && meuEmail() in ['godoygpx@gmail.com']; }

    // Organizador (cliente)
    function orgPath(o) { return /databases/$(database)/documents/organizadores/$(o); }
    function orgExiste(o) { return exists(orgPath(o)); }
    function org(o) { return get(orgPath(o)).data; }
    function orgAtiva(o) { return orgExiste(o) && org(o).status == 'ativo'; }
    function isDono(o) { return logado() && orgExiste(o) && org(o).donoEmail == meuEmail(); }
    // Pode administrar o organizador: você (sempre) ou o dono, com o acesso ativo
    function gestor(o) { return isSuper() || (isDono(o) && orgAtiva(o)); }

    // Equipe de check-in do organizador
    function membroPath(o) { return /databases/$(database)/documents/organizadores/$(o)/equipe/$(meuEmail()); }
    function membro(o) { return get(membroPath(o)).data; }
    function isMembro(o) { return logado() && orgAtiva(o) && exists(membroPath(o)) && membro(o).ativo == true; }
    function membroNoEvento(o, eventoId) {
      return isMembro(o)
        && (membro(o).get('todosEventos', true) == true || eventoId in membro(o).get('eventos', []));
    }
    function membroPodePagar(o) { return isMembro(o) && membro(o).get('podePagar', false) == true; }

    function evento(o, e) { return get(/databases/$(database)/documents/organizadores/$(o)/eventos/$(e)).data; }
    function alterou() { return request.resource.data.diff(resource.data).affectedKeys(); }
    function texto(v, max) { return v is string && v.size() <= max; }

    // Configurações públicas (ex.: organizador dos links antigos)
    match /config/{d} {
      allow read: if true;
      allow write: if isSuper();
    }

    // Login -> organizador (um e-mail pertence a um organizador)
    match /usuarios/{email} {
      allow read: if isSuper() || (logado() && meuEmail() == email)
        || (resource != null && gestor(resource.data.orgId));
      allow create: if isSuper()
        || (request.resource.data.papel == 'equipe' && gestor(request.resource.data.orgId)
            && request.resource.data.keys().hasOnly(['orgId','papel']));
      allow update: if isSuper()
        || (resource.data.papel == 'equipe' && request.resource.data.papel == 'equipe'
            && resource.data.orgId == request.resource.data.orgId && gestor(resource.data.orgId));
      allow delete: if isSuper() || (resource.data.papel == 'equipe' && gestor(resource.data.orgId));
    }

    match /organizadores/{o} {
      // Cadastro e cobrança: só você altera; o dono e a equipe leem (nome e situação)
      allow read: if isSuper() || (logado() && resource.data.donoEmail == meuEmail()) || isMembro(o);
      allow write: if isSuper();

      // Eventos: o público lê enquanto o organizador estiver ativo
      match /eventos/{e} {
        allow read: if orgAtiva(o) || isSuper();
        allow update: if gestor(o);
        // Criar/excluir: você sempre; o dono só se o organizador tiver essa permissão
        allow create, delete: if isSuper() || (gestor(o) && org(o).get('podeCriarEventos', true) == true);
      }

      // Cores e estilo (tema): o público lê para montar a página de inscrição e o voucher
      match /publico/{d} {
        allow read: if orgAtiva(o) || gestor(o);
        allow write: if gestor(o);
      }

      // Histórico de acessos e ações: todos gravam, só você lê; ninguém altera nem apaga (exceto você)
      match /historico/{h} {
        allow read: if isSuper();   // só você vê
        allow create: if (gestor(o) || isMembro(o))
          && request.resource.data.keys().hasOnly(['em','email','nome','papel','acao','texto','disp','eventoId','inscricaoId'])
          && request.resource.data.email == meuEmail()
          && request.resource.data.em == request.time
          && texto(request.resource.data.texto, 300)
          && texto(request.resource.data.acao, 30);
        allow delete: if isSuper();
      }

      // Equipe: o dono gerencia; cada membro lê o próprio cadastro
      match /equipe/{m} {
        allow read: if gestor(o) || (logado() && meuEmail() == m);
        allow write: if gestor(o);
      }

      match /inscricoes/{i} {
        // Abrir um voucher pelo link (precisa do ID exato)
        allow get: if true;
        allow list: if gestor(o) || membroNoEvento(o, resource.data.eventoId);
        allow delete: if gestor(o);

        allow update: if gestor(o)
          // equipe: só registra/desfaz entrada
          || (membroNoEvento(o, resource.data.eventoId)
              && alterou().hasOnly(['checkin','checkinEm','checkinPor','checkinPorNome','checkinLocal']))
          // equipe com permissão de pagamento: também confirma o Pix
          || (membroPodePagar(o) && membroNoEvento(o, resource.data.eventoId)
              && alterou().hasOnly(['checkin','checkinEm','checkinPor','checkinPorNome','checkinLocal','pago','pagoEm','pagoPor','pagoPorNome','valorPago']));

        allow create: if gestor(o) || (
          orgAtiva(o)
          && request.resource.data.keys().hasOnly(['eventoId','codigo','nome','cpf','telefone','email','extra','respostas','grupo','grupoIds','crianca','nascimento','isento','meia','meiaMotivo','termoAceito','termoAceitoEm','termoVersao','termoAceitoPor','checkin','checkinEm','pago','pagoEm','criadoEm'])
          && request.resource.data.eventoId is string
          && evento(o, request.resource.data.eventoId).aberto == true
          && texto(request.resource.data.codigo, 8)
          && texto(request.resource.data.nome, 120) && request.resource.data.nome.size() >= 3
          && texto(request.resource.data.cpf, 11)
          && texto(request.resource.data.telefone, 20)
          && texto(request.resource.data.email, 120)
          && texto(request.resource.data.extra, 120)
          // Campos personalizados (até 20)
          && request.resource.data.get('respostas', {}) is map
          && request.resource.data.get('respostas', {}).size() <= 20
          // Inscrição em grupo (até 10 pessoas)
          && texto(request.resource.data.get('grupo', ''), 20)
          && request.resource.data.get('grupoIds', []) is list
          && request.resource.data.get('grupoIds', []).size() <= 10
          // Termo de responsabilidade: obrigatório quando o evento exige
          && texto(request.resource.data.get('termoAceitoPor', ''), 120)
          && (evento(o, request.resource.data.eventoId).get('termo', {}).get('ativo', false) != true
              || (request.resource.data.get('termoAceito', false) == true
                  && request.resource.data.get('termoAceitoEm', null) == request.time))
          // Meia-entrada: só se o evento aceita
          && texto(request.resource.data.get('meiaMotivo', ''), 80)
          && (request.resource.data.get('meia', false) == false
              || evento(o, request.resource.data.eventoId).get('pix', {}).get('meia', false) == true)
          // Criança isenta: só se o evento oferece gratuidade
          && request.resource.data.get('crianca', false) is bool
          && texto(request.resource.data.get('nascimento', ''), 10)
          && (request.resource.data.get('isento', false) == false
              || (request.resource.data.get('crianca', false) == true
                  && evento(o, request.resource.data.eventoId).get('pix', {}).get('criancasGratis', false) == true))
          && request.resource.data.checkin == false
          && request.resource.data.checkinEm == null
          && request.resource.data.get('pago', false) == false
          && request.resource.data.get('pagoEm', null) == null
          && request.resource.data.criadoEm == request.time
          // Evento que pede CPF: cria junto o bloqueio (impede CPF duplicado); criança pode ficar sem CPF
          && (evento(o, request.resource.data.eventoId).get('campos', {}).get('cpf', false) != true
              || (request.resource.data.get('crianca', false) == true && request.resource.data.cpf == '')
              || existsAfter(/databases/$(database)/documents/organizadores/$(o)/bloqueios/$(request.resource.data.eventoId + '_' + request.resource.data.cpf)))
        );
      }

      // Cobranças Pix automáticas (Mercado Pago): criadas só pelo servidor (Cloud Functions)
      match /cobrancas/{c} {
        allow get: if true;
        allow list: if gestor(o);
        allow write: if false;
      }

      // Bloqueios de CPF por evento: criados (nunca sobrescritos) junto com a inscrição
      match /bloqueios/{b} {
        allow read, update, delete: if gestor(o);
        allow create: if gestor(o) || request.resource.data.keys().hasOnly(['eventoId','inscricaoId'])
          && b == request.resource.data.eventoId + '_' + getAfter(/databases/$(database)/documents/organizadores/$(o)/inscricoes/$(request.resource.data.inscricaoId)).data.cpf
          && getAfter(/databases/$(database)/documents/organizadores/$(o)/inscricoes/$(request.resource.data.inscricaoId)).data.eventoId == request.resource.data.eventoId;
      }
    }

    // mpContas e mpEstados (acessos do Mercado Pago): sem regra = ninguém lê pelo app; só o servidor

    // ===== Dados da versão antiga (só você lê e apaga, para a migração) =====
    match /eventos/{x}    { allow read, delete: if isSuper(); }
    match /inscricoes/{x} { allow read, delete: if isSuper(); }
    match /bloqueios/{x}  { allow read, delete: if isSuper(); }
    match /equipe/{x}     { allow read, delete: if isSuper(); }
  }
}
